import os
import json
import logging
from typing import Dict, Any, Optional

logger = logging.getLogger("AIAnalyzer")



class NetworkAIAnalyzer:
    """
    Motor de anÃ¡lise inteligente com Strategy Pattern, Whitelist flexÃ­vel
    e Blacklist de SeguranÃ§a contra comandos destrutivos.
    """

    # Comandos absolutamente proibidos que a IA JAMAIS pode mandar para um roteador

        "reload",

        "erase",

    def __init__(self, modelo: str = "llama3", ollama_url: str = "http://localhost:11434/api/generate"):
        self.modelo = modelo
        self.ollama_url = ollama_url
        logger.info(f"AI Analyzer inicializado com o modelo '{self.modelo}' (Ollama) e suporte a RAG.")

    def analisar_log_rede(self, log_bruto: str, equipamento: str, historico_recente: Optional[list] = None) -> Dict[str, Any]:
        """
        Envia o log para a IA junto com o histÃ³rico de incidentes recentes recuperados do DynamoDB (RAG).
        """
        # Monta a string de contexto histÃ³rico, se houver registros anteriores
        contexto_historico_str = ""
        if historico_recente and len(historico_recente) > 0:
            contexto_historico_str = "\n".join(
                [f"- Incidente anterior: [{item.get('status')}] {item.get('diagnostico')} (ID: {item.get('incidente_id')})" 
                 for item in historico_recente]
            )
            contexto_historico_str = f"\nHISTÃ“RICO RECENTE DESTE EQUIPAMENTO NAS ÃšLTIMAS HORAS:\n{contexto_historico_str}\n"
        else:
            contexto_historico_str = "\nHISTÃ“RICO RECENTE: Nenhum incidente anterior registrado recentemente (Primeiro evento).\n"

        prompt = f"""
        VocÃª Ã© um Engenheiro SÃªnior de Redes e DevOps especialista em correlaÃ§Ã£o de falhas.
        {contexto_historico_str}
        Analise o log de erro atual abaixo do equipamento '{equipamento}' e retorne APENAS um objeto JSON vÃ¡lido (sem markdown, sem crases ```json) com exatamente estas chaves:
        - "diagnostico": ExplicaÃ§Ã£o tÃ©cnica clara e direta da causa raiz. Considere o histÃ³rico acima para detectar se Ã© um problema recorrente (ex: flap intermitente).
        - "severidade": "BAIXA", "MEDIA", "ALTA" ou "CRITICA".
        - "comando_remediacao": Um comando CLI seguro para mitigar o problema.

        LOG ATUAL DO EQUIPAMENTO:
        {log_bruto}
        """

        payload = {
            "model": self.modelo,
            "prompt": prompt,
            "stream": False,
            "format": "json",
            "options": {
                "temperature": 0.0
            }
        }

        try:
            logger.info(f"Enviando log do equipamento {equipamento} com contexto RAG para a IA...")
            response = requests.post(self.ollama_url, json=payload, timeout=90)
            
            if response.status_code != 200:
                logger.error(f"Erro na API do Ollama: Status {response.status_code}")
                return self._fallback_resposta(log_bruto)

            resposta_json = response.json()
            texto_gerado = resposta_json.get("response", "{}")
            
            diagnostico_dict = json.loads(texto_gerado)
            
            # ValidaÃ§Ã£o de SeguranÃ§a (Whitelist)
            comando_sugerido = diagnostico_dict.get("comando_remediacao", "")
            if not self._validar_comando_seguro(comando_sugerido):
                logger.warning(f"ALERTA DE SEGURANÃ‡A: Comando bloqueado: '{comando_sugerido}'")
                diagnostico_dict["comando_remediacao"] = "show logging (Comando original bloqueado por seguranÃ§a)"

            logger.info("AnÃ¡lise com RAG concluÃ­da e validada com sucesso.")
            return diagnostico_dict

        except requests.exceptions.ConnectionError:
            logger.warning("Ollama nÃ£o estÃ¡ rodando localmente. Usando fallback.")
            return self._fallback_resposta(log_bruto)
        except Exception as e:
            logger.error(f"Erro inesperado ao consultar a IA: {e}")
            return self._fallback_resposta(log_bruto)

class NetmikoRouterDriver(NetworkDriverInterface):
    """
    Driver de ProduÃ§Ã£o/LaboratÃ³rio real com resiliÃªncia, retry e tratamento avanÃ§ado de erros (Pilar 1 e 2).
    """

    def __init__(self, device_type: str, ip: str, username: str, secret: str, port: int = 22, max_retries: int = 3, retry_delay: int = 2):
        if not NETMIKO_DISPONIVEL:
            raise ImportError("A biblioteca 'netmiko' nÃ£o estÃ¡ instalada. Execute: pip install netmiko")
        
        self.device_config = {
            'device_type': device_type,
            'host': ip,
            'username': username,
            'password': secret,
            'secret': secret,
            'port': port,
            'timeout': 10,
            'session_timeout': 15
        }
        self.ip = ip
        self.max_retries = max_retries
        self.retry_delay = retry_delay
        logger.info(f"Driver Netmiko resiliente configurado para o host: {self.ip} (Max retries: {max_retries})")

    def _executar_com_retry(self, func, *args, **kwargs):
        """Executa uma operaÃ§Ã£o com novas tentativas e intervalo crescente (backoff)."""
        tentativa = 0
        delay_atual = self.retry_delay
        while tentativa < self.max_retries:
            tentativa += 1
            try:
                return func(*args, **kwargs)
            except NetmikoTimeoutException as e:
                logger.warning(
                    f"[TENTATIVA {tentativa}/{self.max_retries}] Timeout em {self.ip}. "
                    f"Tentando novamente em {delay_atual}s..."
                )
                if tentativa == self.max_retries:
                    logger.error(
                        f"[FALHA CRÃTICA] Esgotadas {self.max_retries} tentativas em {self.ip}."
                    )
                    raise ConnectionError(f"Timeout persistente com o roteador {self.ip}") from e
                time.sleep(delay_atual)
                delay_atual *= 2
            except NetmikoAuthenticationException:
                logger.error(
                    f"[ERRO DE SEGURANÃ‡A] Falha de autenticaÃ§Ã£o SSH em {self.ip}. "
                    "Tentativas canceladas."
                )
                raise PermissionError(f"Falha de autenticaÃ§Ã£o no roteador {self.ip}")
            except Exception as e:
                logger.error(
                    f"[ERRO INESPERADO] Falha na tentativa {tentativa} com {self.ip}: {str(e)}"
                )
                if tentativa == self.max_retries:
                    raise
                time.sleep(delay_atual)

    def coletar_logs_recentes(self) -> str:
        def _acao():
            with ConnectHandler(**self.device_config) as conn:
                return conn.send_command("show ip bgp summary")
        
        logger.info(f"Iniciando coleta com polÃ­tica de resiliÃªncia em {self.ip}...")
        return self._executar_com_retry(_acao)

    def aplicar_remediacao(self, comando: str) -> bool:
        def _acao():
            with ConnectHandler(**self.device_config) as conn:
                conn.enable()
                output = conn.send_config_set([comando]) if "config" in comando.lower() else conn.send_command(comando)
                logger.info(f"Comando aplicado com sucesso em {self.ip}. SaÃ­da:\n{output}")
                return True

        logger.info(f"Iniciando remediaÃ§Ã£o com polÃ­tica de resiliÃªncia em {self.ip}: '{comando}'")
        try:
            return self._executar_com_retry(_acao)
        except Exception as e:
            logger.error(f"[FALHA NA REMEDIAÃ‡ÃƒO] NÃ£o foi possÃ­vel aplicar o comando em {self.ip}: {str(e)}")
            return False

logger = logging.getLogger("NetworkParser")

class NetworkParser:
    """
    ResponsÃ¡vel por transformar saÃ­das brutas de CLI de roteadores em estruturas limpas utilizando TextFSM.
    """
    
    @staticmethod
    def parsear_texto(template_str: str, raw_output: str) -> list:
        """
        Recebe o conteÃºdo de um template TextFSM em string e a saÃ­da bruta do comando,
        retornando uma lista de dicionÃ¡rios estruturados.
        """
        try:
            template_file = io.StringIO(template_str)
            fsm = TextFSM(template_file)
            resultado = fsm.ParseText(raw_output)
            
            dados_estruturados = [
                dict(zip(fsm.header, linha)) for linha in resultado
            ]
            logger.info(f"Parsing concluÃ­do com sucesso. {len(dados_estruturados)} registros extraÃ­dos.")
            return dados_estruturados
        except Exception as e:
            logger.error(f"Erro ao processar o TextFSM: {str(e)}")
            return []

    @staticmethod
    def parsear_arquivo(template_path: str, raw_output: str) -> list:
        """
        LÃª um arquivo de template TextFSM do disco e aplica na saÃ­da bruta do comando.
        """
        try:
            with open(template_path, "r", encoding="utf-8") as f:
                template_conteudo = f.read()
                
            return NetworkParser.parsear_texto(template_conteudo, raw_output)
        except Exception as e:
            logger.error(f"Erro ao carregar o arquivo de template {template_path}: {str(e)}")
            return []
